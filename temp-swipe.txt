    let startY = 0
    let startX = 0
    let lastX = 0
    let lastT = 0
    let velocity = 0
    let dragging = false
    let tracking = false

    const onTouchStart = (e: TouchEvent) => {
      startX = e.touches[0].clientX
      // Only allow edge swipe back
      if (startX > 40) return

      tracking = true
      startY = e.touches[0].clientY
      lastX = startX
      lastT = performance.now()
      velocity = 0
      dragging = false
    }

    const onTouchMove = (e: TouchEvent) => {
      if (!tracking) return
      const x = e.touches[0].clientX
      const y = e.touches[0].clientY
      const dx = x - startX
      const dy = y - startY

      if (!dragging) {
        if (dx > 10 && Math.abs(dx) > Math.abs(dy)) {
          dragging = true
          screen.classList.add('screen-dragging')
        } else if (Math.abs(dy) > 10) {
          tracking = false
          return
        } else {
          return
        }
      }

      if (e.cancelable) e.preventDefault()
      const now = performance.now()
      velocity = (x - lastX) / Math.max(1, now - lastT)
      lastX = x
      lastT = now

      const offset = Math.max(0, dx)
      screen.style.transform = "translate3d(\px, 0, 0)"
    }

    const onTouchEnd = () => {
      if (!tracking) return
      tracking = false
      if (!dragging) return
      dragging = false
      screen.classList.remove('screen-dragging')

      const offset = lastX - startX
      if (offset > window.innerWidth / 3 || velocity > 0.5) {
        haptic('light')
        screen.style.transition = 'transform 260ms cubic-bezier(0.2, 0.8, 0.25, 1), opacity 260ms ease'
        screen.style.transform = "translate3d(100%, 0, 0)"
        screen.style.opacity = '0'
        window.setTimeout(() => {
          resetTabbar()
          nav.back()
        }, 260)
      } else {
        screen.style.transition = 'transform 240ms cubic-bezier(0.25, 1, 0.5, 1)'
        screen.style.transform = ''
        window.setTimeout(() => {
          if (screen) screen.style.transition = ''
        }, 250)
      }
    }
